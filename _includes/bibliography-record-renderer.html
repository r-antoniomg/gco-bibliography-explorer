For articles and chapters, pages usually indicates the location of the contribution within a larger work. For books, it may represent extent or physical description. The shared renderer therefore displays pages only for item types where pages are likely to be a page range or location within a container. */ const ITEM_TYPES_WITH_PAGE_RANGES = [ 'journalArticle', 'bookSection', 'conferencePaper', 'magazineArticle', 'newspaperArticle', 'encyclopediaArticle', 'dictionaryEntry', 'report', 'thesis' ]; function escapeHtml(value) { if (value === null || value === undefined) { return ''; } return String(value) .replace(/&/g, '&') .replace(//g, '>') .replace(/"/g, '"'); } function normalizeZoteroRecord(item) { const record = item.data || {}; return { zoteroKey: item.key || record.key || '', itemType: record.itemType || '', title: record.title || '', creators: record.creators || [], publicationTitle: record.publicationTitle || '', bookTitle: record.bookTitle || '', volume: record.volume || '', issue: record.issue || '', pages: record.pages || '', place: record.place || '', publisher: record.publisher || '', date: record.date || '', language: record.language || '', url: record.url || '', doi: record.DOI || record.doi || '', isbn: record.ISBN || record.isbn || '', issn: record.ISSN || record.issn || '', tags: record.tags || [], abstract: record.abstractNote || '' }; } function hasRenderableValue(value) { if (Array.isArray(value)) { return value.length > 0; } return value !== null && value !== undefined && String(value).trim() !== ''; } function shouldDisplayPages(normalizedRecord) { return ITEM_TYPES_WITH_PAGE_RANGES.includes(normalizedRecord.itemType); } function formatCreators(creators) { if (!creators || creators.length === 0) { return ''; } return creators .map(creator => { if (creator.name) { return escapeHtml(creator.name); } return [creator.firstName, creator.lastName] .filter(Boolean) .map(escapeHtml) .join(' '); }) .filter(Boolean) .join('; '); } function formatUrl(url) { if (!hasRenderableValue(url)) { return ''; } const safeUrl = escapeHtml(url); return `

${RECORD_FIELD_LABELS.url}: ${safeUrl}

`; } function formatTags(tags) { if (!tags || tags.length === 0) { return ''; } const tagButtons = tags .map(tagObject => { const tag = tagObject.tag || ''; if (!tag) { return ''; } const encodedTag = encodeURIComponent(tag); const safeTag = escapeHtml(tag); return ``; }) .filter(Boolean) .join(' '); if (!tagButtons) { return ''; } return `

Explore this topic across all collections:
${tagButtons}

`; } function formatSimpleField(fieldName, value) { if (!hasRenderableValue(value)) { return ''; } const label = RECORD_FIELD_LABELS[fieldName] || fieldName; return `

${escapeHtml(label)}: ${escapeHtml(value)}

`; } function renderRecordField(fieldName, normalizedRecord) { if (fieldName === 'title') { return ''; } if (fieldName === 'zoteroKey' && !SHOW_ZOTERO_KEY) { return ''; } if (fieldName === 'pages' && !shouldDisplayPages(normalizedRecord)) { return ''; } const value = normalizedRecord[fieldName]; if (!hasRenderableValue(value)) { return ''; } if (fieldName === 'creators') { const creators = formatCreators(value); return creators ? formatSimpleField('creators', creators) : ''; } if (fieldName === 'url') { return formatUrl(value); } if (fieldName === 'tags') { return formatTags(value); } if (fieldName === 'abstract') { return `

${RECORD_FIELD_LABELS.abstract}: ${escapeHtml(value)}

`; } return formatSimpleField(fieldName, value); } function renderBibliographyRecord(item) { const normalizedRecord = normalizeZoteroRecord(item); const displayFields = Array.isArray(RECORD_DISPLAY_FIELDS) ? RECORD_DISPLAY_FIELDS : String(RECORD_DISPLAY_FIELDS).split(',').map(field => field.trim()).filter(Boolean); let html = `

${escapeHtml(normalizedRecord.title || 'Untitled')}

`; displayFields.forEach(fieldName => { html += renderRecordField(fieldName, normalizedRecord); }); html += `
`; return html; }
